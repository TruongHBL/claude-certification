1. Mục tiêu

Model chọn tool chủ yếu dựa vào name + description. Bài này tái hiện lỗi misrouting khi hai tool có mô tả mơ hồ, chồng lấn; sửa bằng mô tả đủ 5 yếu tố (mục đích, input, ví dụ, giới hạn, ranh giới); rồi tìm câu nhạy từ khoá trong system prompt có thể đè lên mô tả tốt. Ba cấu hình A/B/C tách riêng tác động của mô tả và của system prompt, chấm tự động theo nhãn.

export const TOOLS = TOOLSET === "clear" ? CLEAR_TOOLS : VAGUE_TOOLS;

type Args = Record<string, unknown>;
const joinArgs = (args: Args, keys: string[]) =>
  keys.map((k) => args[k]).filter((v) => v != null).map(String).join(" ");

// Ở bản vague, field `query` là chuỗi tự do: dò mọi kiểu định danh trong chuỗi.
function findCustomer(args: Args) {
  const q = joinArgs(args, ["customer_id", "email", "phone", "query"]);
  return CUSTOMERS.find((c) =>
    q.includes(c.customer_id) || q.toLowerCase().includes(c.email) || q.replace(/\D/g, "").includes(c.phone));
}

function findOrder(args: Args) {
  const q = joinArgs(args, ["order_number", "tracking_id", "query"]);
  return ORDERS.find((o) => q.includes(o.order_number) || (o.tracking_id != null && q.includes(o.tracking_id)));
}

function respond(found: object | undefined, args: Args): ToolResult {
  return found ? textResult(found) : textResult({ error: "not_found", input: args }, { isError: true });
}

runServer({
  name: "shop",
  tools: TOOLS,
  onCall: (tool, args) => {
    if (LOG_FILE) {
      appendFileSync(LOG_FILE, JSON.stringify({ at: new Date().toISOString(), toolset: TOOLSET, tool, arguments: args }) + "\n");
    }
  },
  handlers: {
    get_customer: (args) => respond(findCustomer(args), args),
    lookup_order: (args) => respond(findOrder(args), args),
  },
});

#!/usr/bin/env bash
# Bài 2.1 – chạy 10 query qua Claude Code CLI với 3 cấu hình rồi chấm tool được gọi ĐẦU TIÊN.
#
#   A = mô tả vague + system prompt conflict   (tình trạng ban đầu)
#   B = mô tả clear + system prompt conflict   (chỉ sửa mô tả)
#   C = mô tả clear + system prompt fixed      (sửa cả mô tả lẫn system prompt)
#
# Dùng:  ./run.sh A | B | C | all     chạy (MỖI query một lệnh `claude -p` riêng, không chung context)
#        ./run.sh score               in bảng độ chính xác + danh sách câu sai
#        ./run.sh clean               xoá results/
#
# Chi phí ước tính (haiku): ~10–20k token input mỗi lệnh (system prompt CLI + mô tả tool), 1–2 lượt.
# 30 lệnh ≈ 0,3–0,8 USD. Mỗi lệnh bị chặn trần bằng --max-budget-usd.
set -euo pipefail

ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
RESULTS="${RESULTS_DIR:-$ROOT/results}"   # test đặt RESULTS_DIR riêng để không ghi đè kết quả thật
NODE="${NODE_BIN:-node}"
TS=(--experimental-strip-types --no-warnings)
MODEL="${MODEL:-haiku}"
BUDGET="${BUDGET:-0.2}"
CLAUDE_BIN="${CLAUDE_BIN:-claude}"

toolset_of() { case "$1" in A) echo vague ;; B|C) echo clear ;; esac; }
prompt_of()  { case "$1" in A|B) echo "$ROOT/prompts/system_conflict.txt" ;; C) echo "$ROOT/prompts/system_fixed.txt" ;; esac; }

run_config() {
  local cfg="$1" toolset prompt dir n
  toolset="$(toolset_of "$cfg")"; prompt="$(prompt_of "$cfg")"
  dir="$RESULTS/$cfg"; mkdir -p "$dir"
  n="$("$NODE" -e 'console.log(require(process.argv[1]).queries.length)' "$ROOT/queries.json")"
  echo "== Cấu hình $cfg: TOOLSET=$toolset, prompt=$(basename "$prompt"), model=$MODEL"
  for ((i = 0; i < n; i++)); do
    local id text mcp
    id="$("$NODE" -e 'console.log(require(process.argv[1]).queries[+process.argv[2]].id)' "$ROOT/queries.json" "$i")"
    text="$("$NODE" -e 'console.log(require(process.argv[1]).queries[+process.argv[2]].text)' "$ROOT/queries.json" "$i")"
    mcp="$dir/q$id.mcp.json"
    rm -f "$dir/q$id.log"
    # Mỗi query một file cấu hình MCP để log tool call tách riêng theo query.
    # Server chạy bằng node (đường dẫn tuyệt đối) với --experimental-strip-types.
    "$NODE" -e 'const [srv, toolset, log, node] = process.argv.slice(1);
      console.log(JSON.stringify({ mcpServers: { shop: { type: "stdio", command: node,
        args: ["--experimental-strip-types", "--no-warnings", srv], env: { TOOLSET: toolset, SHOP_LOG: log } } } }, null, 2))' \
      "$ROOT/shop_mcp_server.ts" "$toolset" "$dir/q$id.log" "$(command -v "$NODE")" > "$mcp"
    echo "  q$id: $text"
    # Chạy trong thư mục rỗng để không nạp CLAUDE.md/settings của project nào.
    (cd "$RESULTS" && "$CLAUDE_BIN" -p "$text" \
      --model "$MODEL" \
      --system-prompt "$(cat "$prompt")" \
      --mcp-config "$mcp" --strict-mcp-config \
      --tools "" \
      --allowedTools "mcp__shop__get_customer" "mcp__shop__lookup_order" \
      --permission-mode dontAsk \
      --max-budget-usd "$BUDGET" \
      --no-session-persistence \
      --output-format stream-json --verbose \
      > "$dir/q$id.jsonl" 2> "$dir/q$id.err") || echo "    (lệnh trả mã lỗi, xem $dir/q$id.err)"
  done
}

case "${1:-}" in
  A|B|C) run_config "$1" ;;
  all)   for c in A B C; do run_config "$c"; done; "$NODE" "${TS[@]}" "$ROOT/score.ts" "$RESULTS" "$ROOT/queries.json" ;;
  score) "$NODE" "${TS[@]}" "$ROOT/score.ts" "$RESULTS" "$ROOT/queries.json" ;;
  clean) rm -rf "$RESULTS" ;;
  *) echo "Dùng: $0 A|B|C|all|score|clean"; exit 1 ;;
esac

File score.ts

#!/usr/bin/env -S node --experimental-strip-types --no-warnings
// Bài 2.1 – chấm tool được gọi ĐẦU TIÊN cho từng query, từng cấu hình.
// Nguồn chính: block tool_use đầu tiên trong stream-json (qN.jsonl).
// Dự phòng: dòng đầu của log server (qN.log), nếu stream-json không đọc được.
// Dùng: node --experimental-strip-types score.ts results queries.json
import { existsSync, readFileSync, statSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { readStream, toolUses } from "../lib/cli_stream.ts";

interface Query { id: number; text: string; expected: string }

const [resultsDir, queriesFile] = process.argv.slice(2);
if (!resultsDir || !queriesFile) {
  console.error("Dùng: score.ts <results_dir> <queries.json>");
  process.exit(2);
}
const queries = (JSON.parse(readFileSync(queriesFile, "utf8")) as { queries: Query[] }).queries;
const CONFIGS: Record<string, string> = { A: "vague + conflict", B: "clear + conflict", C: "clear + fixed" };

function firstToolFromStream(path: string): string | undefined {
  return toolUses(readStream(path))[0]?.name.replace(/^mcp__shop__/, "");
}

function firstToolFromLog(path: string): string | undefined {
  if (!existsSync(path) || statSync(path).size === 0) return undefined;
  const first = readFileSync(path, "utf8").split("\n")[0];
  return (JSON.parse(first) as { tool: string }).tool;
}

const summary: { config: string; label: string; correct: number; total: number }[] = [];
for (const [cfg, label] of Object.entries(CONFIGS)) {
  const dir = join(resultsDir, cfg);
  if (!existsSync(dir)) continue;

  const rows = queries.map((q) => {
    const got = firstToolFromStream(join(dir, `q${q.id}.jsonl`)) ??
      firstToolFromLog(join(dir, `q${q.id}.log`)) ?? "(không gọi tool)";
    return { ...q, got, ok: got === q.expected };
  });
  const correct = rows.filter((r) => r.ok).length;
  summary.push({ config: cfg, label, correct, total: rows.length });
  console.log(`\n== ${cfg} (${label}): ${correct}/${rows.length}`);
  for (const r of rows.filter((r) => !r.ok)) {
    console.log(`   ✗ q${r.id}: cần ${r.expected}, gọi ${r.got}  | ${r.text}`);
  }
}

if (summary.length === 0) {
  console.log(`Chưa có kết quả trong ${resultsDir}. Chạy ./run.sh all trước.`);
  process.exit(1);
}
console.log("\nCấu hình             Đúng");
for (const s of summary) {
  console.log(`${s.config.padEnd(2)} ${s.label.padEnd(17)} ${String(s.correct).padStart(2)}/${s.total}`);
}
writeFileSync(join(resultsDir, "score.json"), JSON.stringify(summary, null, 2) + "\n");

10. Tiêu chí hoàn thành

  • Có so sánh độ chính xác trước/sau trên cùng 10 query (A → B → C, chấm tự động bằng score.ts)
  • Mô tả mới nêu định dạng định danh (#NNNNN, CUS-NNNN, mã vận đơn, SĐT 10 số) và ranh giới “KHÔNG dùng cho…” giữa hai tool
  • Có danh sách câu system prompt gây xung đột kèm bản viết lại (mục 5)
  • Mục tiêu khi chạy thật: C đạt 9–10/10. Đạt: 10/10 ở cả 2 lượt

11. Bài học rút ra

  • Description là cơ chế chọn tool chính. Mô tả một câu chồng lấn thì model đoán theo từ khoá. Sửa mô tả là bước đầu tiên: rẻ và nhắm thẳng nguyên nhân.
  • Mô tả tốt có 5 yếu tố: mục đích, định dạng input, ví dụ query, edge case/giới hạn, ranh giới với tool tương tự. Schema tách field có pattern giúp thêm, nhưng không thay được ranh giới viết bằng lời.
  • Bẫy đề thi (đều sai khi hỏi “bước đầu tiên”): thêm few-shot vào system prompt (tốn token, không sửa gốc), routing classifier theo từ khoá (quá nặng, bỏ qua khả năng hiểu ngôn ngữ của model), gộp tool thành một (hợp lệ về lâu dài nhưng tốn công hơn).
  • Sửa mô tả rồi phải đọc lại system prompt. Câu kiểu “luôn kiểm tra thông tin khách hàng” tạo liên kết từ khoá với một tool và đè lên mô tả tốt (cấu hình B vs C).
  • Tách tool khi tool quá chung (vd analyze_document → extract_data_points, summarize_content, verify_claim_against_source); đổi tên khi tên dễ nhầm (vd analyze_content → extract_web_results).
  • Giới hạn: mô tả chỉ chữa nhầm giữa một số ít tool. Khi một agent có quá ~4–5 tool, chọn sai vì số lượng; cách sửa là phân bổ tool theo vai trò (bài 2.3).
