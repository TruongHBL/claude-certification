1. Mục tiêu

Cấu hình một hệ coordinator + 3 subagent (web_search, doc_analysis, synthesis), mỗi agent 4–5 tool đúng vai trò. Thêm một tool cross-role có giới hạn (verify_fact) để tránh vòng về coordinator, dùng tool_choice ép thứ tự bước bắt buộc, thay fetch_url bằng load_document có kiểm tra URL, rồi chạy end-to-end và kiểm chứng từ log rằng không agent nào dùng tool ngoài bộ của mình.

Bài dùng Messages API trực tiếp (TypeScript, gọi bằng fetch qua lib/anthropic.ts, không cần SDK) vì tool_choice là tham số của Messages API và phải đổi theo từng lượt (doc_analysis: lượt 1 ép extract_metadata, sau đó auto). Agent SDK / Claude Code không cho đặt tool_choice từng lượt như vậy.

// load_document: bản THU HẸP của fetch_url (least privilege áp vào thiết kế tool).

import { isIP } from "node:net";
import { extname } from "node:path";

export const DOC_EXTENSIONS = [".pdf", ".docx", ".md", ".txt", ".html", ".htm"];
export const TRUSTED_DOMAINS = ["gso.gov.vn", "molisa.gov.vn", "ilo.org", "worldbank.org", "vnexpress.net", "hbr.org", "docs.example.vn"];
export const BLOCKED_HOSTS = ["localhost", "metadata.google.internal"];

export interface StructuredError {
  errorCategory: "transient" | "validation" | "business" | "permission";
  isRetryable: boolean;
  description: string;
}

export interface LoadDocumentError {
  isError: true;
  structuredContent: StructuredError;
}

function error(category: StructuredError["errorCategory"], description: string): LoadDocumentError {
  return { isError: true, structuredContent: { errorCategory: category, isRetryable: false, description } };
}

/** Loopback, private (10/8, 172.16/12, 192.168/16, fc00::/7) hoặc link-local (169.254/16, fe80::/10). */
export function isInternalIp(host: string): boolean {
  const h = host.replace(/^\[|\]$/g, "");
  const v = isIP(h);
  if (v === 4) {
    const [a, b] = h.split(".").map(Number);
    return a === 127 || a === 10 || (a === 172 && b >= 16 && b <= 31) || (a === 192 && b === 168) || (a === 169 && b === 254);
  }
  if (v === 6) {
    const l = h.toLowerCase();
    return l === "::1" || /^f[cd]/.test(l) || /^fe[89ab]/.test(l);
  }
  return false;
}

const trusted = (host: string) => TRUSTED_DOMAINS.some((d) => host === d || host.endsWith(`.${d}`));

/** Trả null nếu hợp lệ, hoặc object lỗi có cấu trúc. */
export function validate(url: unknown): LoadDocumentError | null {
  const raw = String(url ?? "").trim();
  let u: URL;
  try {
    u = new URL(raw);
  } catch {
    return error("validation", `URL không hợp lệ: '${raw}'. Cần URL https đầy đủ trỏ tới tài liệu.`);
  }
  if (!u.hostname) return error("validation", `URL không hợp lệ: '${raw}'. Cần URL https đầy đủ trỏ tới tài liệu.`);

  const host = u.hostname.toLowerCase();
  const scheme = u.protocol.replace(/:$/, "");
  if (scheme !== "https") return error("validation", `Chỉ nhận https (nhận được ${scheme}). Tìm bản https của tài liệu.`);
  if (BLOCKED_HOSTS.includes(host) || isInternalIp(host)) {
    return error("permission", `Từ chối địa chỉ nội bộ '${host}'. load_document không truy cập mạng nội bộ/metadata; không thử lại.`);
  }
  if (!trusted(host)) return error("permission", `Domain '${host}' chưa được duyệt. Báo coordinator nếu thật sự cần nguồn này.`);

  const ext = extname(u.pathname).toLowerCase();
  if (!DOC_EXTENSIONS.includes(ext)) {
    const shown = ext === "" ? "(không có đuôi)" : ext;
    return error("validation", `'${shown}' không phải tài liệu. Chỉ nhận ${DOC_EXTENSIONS.join(" ")}. Dùng search_web để tìm bản tài liệu.`);
  }
  return null;
}

// Payload: tool_choice theo lượt + ràng buộc API
// ---------------------------------------------------------------------------
export function buildPayload(
  agent: Role, turn: number, messages: Message[],
  { thinking }: { thinking?: MessagesRequest["thinking"] } = {},
): MessagesRequest & { tool_choice: ToolChoice } {
  const cfg = AGENTS[agent];
  const choice = cfg.tool_choice(turn);
  // Ràng buộc của Messages API: khi bật extended thinking, tool_choice chỉ được auto hoặc none.
  if (thinking && (choice.type === "any" || choice.type === "tool")) {
    throw new RangeError(`tool_choice ${choice.type} không dùng được cùng extended thinking (chỉ auto/none)`);
  }
  const payload: MessagesRequest & { tool_choice: ToolChoice } = {
    model: MODEL, max_tokens: 2048, system: SYSTEM_PROMPTS[agent],
    tools: cfg.tools, tool_choice: choice, messages,
  };
  if (thinking) payload.thinking = thinking;
  return payload;
}

  /** Trả [kết quả text hoặc input của terminal tool, session_id] */
  async runAgent(agent: Role, prompt: string, { sessionTag }: { sessionTag?: string } = {}): Promise<[string | Json, string]> {
    const n = (this.sessions.get(agent) ?? 0) + 1;
    this.sessions.set(agent, n);
    const session = `${agent}#${n}${sessionTag ? `-${sessionTag}` : ""}`;
    const cfg = AGENTS[agent];
    const allowed = allowedNames(agent);
    const messages: Message[] = [{ role: "user", content: prompt }];

    for (let turn = 1; turn <= MAX_TURNS; turn++) {
      const payload = buildPayload(agent, turn, messages);
      const resp = await this.client.create(payload, { agent, session });
      const uses = resp.content.filter((b): b is ToolUse => b.type === "tool_use");
      this.record({
        agent, session, turn, tool_choice: payload.tool_choice,
        tools_offered: allowed, stop_reason: resp.stop_reason, tool_calls: uses.map((u) => u.name),
      });
      messages.push({ role: "assistant", content: resp.content });

      if (resp.stop_reason !== "tool_use") {
        return [resp.content.flatMap((b) => (b.type === "text" ? [b.text] : [])).join(""), session];
      }

      let terminalInput: Json | null = null;
      const results: UserBlock[] = [];
      for (const u of uses) {
        const [content, isErr] = await this.dispatch(agent, session, allowed, u);
        if (isErr && allowed.includes(u.name)) {
          this.record({ type: "tool_error", agent, session, tool: u.name, error: content });
        }
        if (u.name === cfg.terminal_tool && !isErr) terminalInput = u.input;
        results.push({ type: "tool_result", tool_use_id: u.id, content: JSON.stringify(content), is_error: isErr });
      }
      messages.push({ role: "user", content: results });
      if (terminalInput) return [terminalInput, session];
    }
    throw new Error(`${session} vượt ${MAX_TURNS} lượt`);
  }

  async dispatch(agent: Role, session: string, allowed: string[], use: Pick<ToolUse, "name" | "input">): Promise<ToolOutcome> {
    const { name } = use;
    if (!allowed.includes(name)) {
      this.violations.push({ agent, session, tool: name });
      this.record({ type: "violation", agent, session, tool: name });
      return [{
        errorCategory: "permission", isRetryable: false,
        description: `Tool '${name}' không thuộc bộ tool của ${agent}. Dùng tool của bạn hoặc trả việc về coordinator.`,
      }, true];
    }
    if (name === "Agent") {
      const sub = use.input.agent_type as Role;
      const prompt = String(use.input.prompt);
      const tag = prompt.startsWith("Kiểm chứng") ? "verify" : undefined;
      const [out] = await this.runAgent(sub, prompt, { sessionTag: tag });
      return [typeof out === "string" ? { result: out } : { report: out }, false];
    }
    return this.world.execute(name, use.input);
  }

  async run(question = QUESTION): Promise<string> {
    const [final] = await this.runAgent("coordinator", question);
    return String(final);
  }
}

export function verifyLog(runner: Runner): [LogCheck[], Record<string, string[]>] {

  const checks: LogCheck[] = [];

  for (const e of calls) (used[e.agent] ??= []).push(...e.tool_calls);
  const outside = Object.entries(used).flatMap(([a, names]) =>
    [...new Set(names)].filter((n) => !allowedNames(a).includes(n)).map((n) => `${a}→${n}`));
  add("mỗi agent chỉ dùng tool trong bộ của mình", outside.length === 0, outside.length === 0 ? "OK" : outside.join(", "));

  const docSessions = new Map<string, CallEntry[]>();
  for (const e of calls.filter((e) => e.agent === "doc_analysis")) docSessions.set(e.session, [...(docSessions.get(e.session) ?? []), e]);
  const firstOk = [...docSessions.values()].every((es) => {
    const f = es.reduce((a, b) => (b.turn < a.turn ? b : a));
    return f.tool_choice.type === "tool" && f.tool_choice.name === "extract_metadata" &&
      f.tool_calls.join() === "extract_metadata" &&
      es.filter((e) => e.turn !== 1).every((e) => e.tool_choice.type === "auto");
  });
  add(`doc_analysis: lượt 1 forced extract_metadata, các lượt sau auto (${docSessions.size} phiên)`, firstOk);

  const syn = used.synthesis ?? [];
  add("synthesis dùng compile_report + verify_fact", syn.includes("compile_report") && syn.includes("verify_fact"),
    Object.entries(tally(syn)).map(([k, v]) => `${k}×${v}`).join(", "));
  add("synthesis luôn chạy với tool_choice any",
    calls.filter((e) => e.agent === "synthesis").every((e) => e.tool_choice.type === "any"));

  const routed = [...docSessions.keys()].some((s) => s.endsWith("-verify")) && (used.doc_analysis ?? []).includes("verify_claim");
  add("nhận định phức tạp được chuyển coordinator → doc_analysis verify_claim", routed);
  add("không có vi phạm audit", runner.violations.length === 0, `${runner.violations.length} vi phạm`);
  return [checks, used];
}

  • Số tool là quyết định kiến trúc. Mỗi agent 4–5 tool đúng vai trò. Đúng vai trò quan trọng ngang số lượng: synthesis có search_web sẽ tự đi tìm thay vì dùng kết quả đã được giao.
  • Tool cross-role có giới hạn cho việc đơn giản, tần suất cao: 85% tra cứu đơn giản xử lý tại chỗ, 15% phức tạp vẫn qua coordinator. Không bỏ bước kiểm chứng, không route mọi thứ qua coordinator.
  • tool_choice là công cụ đảm bảo bằng code (giống bài 1.4): forced đảm bảo thứ tự bước, any đảm bảo có output có cấu trúc. Prompt “hãy lấy metadata trước” chỉ là xác suất.
  • Least privilege áp vào thiết kế tool: tool hẹp + kiểm tra đầu vào bằng code, lỗi trả theo quy ước bài 2.2 để agent biết không thử lại.

Bẫy đề

  1. Route mọi kiểm chứng đơn giản qua coordinator khi 85% là tra cứu đơn giản → thêm 2–3 vòng, chậm ~40%. Đúng: verify_fact scoped trên synthesis.
  2. Dùng auto khi bắt buộc cần output có cấu trúc → model có thể trả text. Đúng: any hoặc forced.
  3. Cho một agent 18 tool mà mong chọn đúng → độ tin cậy giảm theo số tool. Đúng: 4–5 tool/agent.
  4. Cho subagent fetch_url chung chung khi load_document là đủ → mở đường lạm dụng.

Câu luyện tập

Synthesis agent thường trả quyền về coordinator để kiểm chứng dữ kiện đơn giản, mỗi việc thêm 2–3 vòng và tăng 40% độ trễ. 85% lần kiểm chứng là tra cứu đơn giản. Cách nào tốt nhất?

  1. Cho synthesis một tool verify_fact scoped cho tra cứu đơn giản, chỉ route kiểm chứng phức tạp qua coordinator.
  2. Bỏ hẳn bước kiểm chứng khỏi synthesis.
  3. Cache mọi kết quả kiểm chứng ở coordinator.
  4. Tăng song song ở coordinator.

Đáp án A. B bỏ mất năng lực cần thiết. C lần đầu vẫn phải đi vòng và không giải quyết mẫu 85%. D chỉ giảm hàng đợi, không bớt số vòng.
