Người thực hiện: TruongNQ · Ngày nộp: 13/10/2026

1. Mục tiêu

Cấu hình một hệ coordinator + 3 subagent (web_search, doc_analysis, synthesis), mỗi agent 4–5 tool đúng vai trò. Thêm một tool cross-role có giới hạn (verify_fact) để tránh vòng về coordinator, dùng tool_choice ép thứ tự bước bắt buộc, thay fetch_url bằng load_document có kiểm tra URL, rồi chạy end-to-end và kiểm chứng từ log rằng không agent nào dùng tool ngoài bộ của mình.

Bài dùng Messages API trực tiếp (Ruby, Net::HTTP, không cần gem) vì tool_choice là tham số của Messages API và phải đổi theo từng lượt (doc_analysis: lượt 1 ép extract_metadata, sau đó auto). Agent SDK / Claude Code không cho đặt tool_choice từng lượt như vậy.

# load_document: bản THU HẸP của fetch_url (least privilege áp vào thiết kế tool).
#
# fetch_url tải BẤT KỲ URL nào → agent có thể gọi API nội bộ, tải ảnh/binary, chạm metadata cloud.
# load_document chỉ nhận tài liệu: https + domain đã duyệt + đuôi tài liệu. Vi phạm trả lỗi có
# cấu trúc theo quy ước bài 2.2 (errorCategory / isRetryable / description).

require "uri"
require "ipaddr"

module LoadDocument
  DOC_EXTENSIONS = %w[.pdf .docx .md .txt .html .htm].freeze
  TRUSTED_DOMAINS = %w[gso.gov.vn molisa.gov.vn ilo.org worldbank.org vnexpress.net hbr.org docs.example.vn].freeze
  BLOCKED_HOSTS = %w[localhost metadata.google.internal].freeze

  module_function

  def error(category, description)
    { isError: true, structuredContent: { errorCategory: category, isRetryable: false, description: description } }
  end

  def internal_ip?(host)
    ip = IPAddr.new(host)
    ip.loopback? || ip.private? || ip.link_local?
  rescue IPAddr::InvalidAddressError
    false
  end

  def trusted?(host)
    TRUSTED_DOMAINS.any? { |d| host == d || host.end_with?(".#{d}") }
  end

  # Trả nil nếu hợp lệ, hoặc Hash lỗi có cấu trúc.
  def validate(url)
    uri = URI.parse(url.to_s.strip)
    return error("validation", "URL không hợp lệ: '#{url}'. Cần URL https đầy đủ trỏ tới tài liệu.") unless uri.is_a?(URI::Generic) && uri.host

    host = uri.host.downcase
    return error("validation", "Chỉ nhận https (nhận được #{uri.scheme}). Tìm bản https của tài liệu.") unless uri.scheme == "https"
    if BLOCKED_HOSTS.include?(host) || internal_ip?(host)
      return error("permission", "Từ chối địa chỉ nội bộ '#{host}'. load_document không truy cập mạng nội bộ/metadata; không thử lại.")
    end
    return error("permission", "Domain '#{host}' chưa được duyệt. Báo coordinator nếu thật sự cần nguồn này.") unless trusted?(host)

    ext = File.extname(uri.path.to_s).downcase
    unless DOC_EXTENSIONS.include?(ext)
      shown = ext.empty? ? "(không có đuôi)" : ext
      return error("validation", "'#{shown}' không phải tài liệu. Chỉ nhận #{DOC_EXTENSIONS.join(' ')}. Dùng search_web để tìm bản tài liệu.")
    end
    nil
  rescue URI::InvalidURIError
    error("validation", "URL không parse được: '#{url}'.")
  end
end

require_relative "tool_config"

module ConfigCheck
  GENERIC_NAMES = %w[fetch_url analyze analyze_document analyze_content process run do_task
                     execute get_data handle helper tool search].freeze
  SUBAGENTS = %w[web_search doc_analysis synthesis].freeze

  module_function

  def checks(agents = ToolConfig::AGENTS, scoped = ToolConfig::SCOPED_CROSS_ROLE)
    out = []
    add = ->(name, ok, detail) { out << { name: name, ok: ok, detail: detail } }

    SUBAGENTS.each do |a|
      n = agents[a][:tools].size
      add.("#{a}: 4–5 tool", (4..5).cover?(n), "#{n} tool")
    end

    owners = Hash.new { |h, k| h[k] = [] }
    agents.each { |a, cfg| cfg[:tools].each { |t| owners[t[:name]] << a } }
    dup = owners.select { |name, list| list.size > 1 && !scoped.key?(name) }
    add.("không tool nào trùng giữa các role (trừ scoped)", dup.empty?, dup.empty? ? "OK" : dup.inspect)

    scoped.each do |name, meta|
      owner_ok = owners[name] == [meta[:owner]]
      desc = agents[meta[:owner]][:tools].find { |t| t[:name] == name }&.dig(:description).to_s
      add.("scoped #{name} chỉ ở #{meta[:owner]} và mô tả có giới hạn",
           owner_ok && desc.include?("[SCOPED]") && desc.include?("coordinator"),
           "owner=#{owners[name].inspect}")
    end

    all_names = owners.keys
    generic = all_names & GENERIC_NAMES
    add.("không có tên tool chung chung / fetch_url", generic.empty?, generic.empty? ? "OK" : generic.inspect)

    domain = SUBAGENTS.flat_map { |a| agents[a][:tools].map { |t| t[:name] } }
    leaked = agents["coordinator"][:tools].map { |t| t[:name] } & domain
    add.("coordinator không giữ tool nghiệp vụ", leaked.empty?, leaked.empty? ? "OK" : leaked.inspect)

    missing_desc = all_names.select { |n| agents.values.flat_map { |c| c[:tools] }.find { |t| t[:name] == n }[:description].to_s.size < 40 }
    add.("mọi tool có mô tả ≥ 40 ký tự", missing_desc.empty?, missing_desc.empty? ? "OK" : missing_desc.inspect)

    first = agents["doc_analysis"][:tool_choice].call(1)
    second = agents["doc_analysis"][:tool_choice].call(2)
    add.("doc_analysis: lượt 1 ép extract_metadata, lượt 2 auto",
         first == { type: "tool", name: "extract_metadata" } && second == { type: "auto" },
         "#{first.inspect} → #{second.inspect}")
    out
  end
end

  failed = rows.count { |r| !r[:ok] }
  puts "\n#{rows.size - failed}/#{rows.size} đạt"
  exit(failed.zero? ? 0 : 1)

def build_payload(agent, turn, messages, thinking: nil)
  cfg = ToolConfig::AGENTS.fetch(agent)
  choice = cfg[:tool_choice].call(turn)
  # Ràng buộc của Messages API: khi bật extended thinking, tool_choice chỉ được auto hoặc none.
  if thinking && %w[any tool].include?(choice[:type])
    raise ArgumentError, "tool_choice #{choice[:type]} không dùng được cùng extended thinking (chỉ auto/none)"
  end

  payload = { model: MODEL, max_tokens: 2048, system: SYSTEM_PROMPTS.fetch(agent),
              tools: cfg[:tools], tool_choice: choice, messages: messages }
  payload[:thinking] = thinking if thinking
  payload
end

class RealClient
  URL = URI("https://api.anthropic.com/v1/messages")
  def initialize(key) = @key = key

  def create(payload, agent:, session:)
    req = Net::HTTP::Post.new(URL, "content-type" => "application/json", "x-api-key" => @key,
                                   "anthropic-version" => "2023-06-01")
    req.body = JSON.generate(payload)
    res = Net::HTTP.start(URL.host, URL.port, use_ssl: true, read_timeout: 120) { |h| h.request(req) }
    raise "HTTP #{res.code}: #{res.body[0, 300]}" unless res.is_a?(Net::HTTPSuccess)
    JSON.parse(res.body)
  end
end

# ---------------------------------------------------------------------------
# Runner: vòng lặp agent + audit allowlist + log JSONL
# ---------------------------------------------------------------------------
class Runner
  attr_reader :log, :violations, :world

  def initialize(client, log_path: nil)
    @client = client
    @world = MockWorld.new
    @log = []
    @violations = []
    @sessions = Hash.new(0)
    @log_path = log_path
  end

  def record(entry)
    @log << entry
    File.open(@log_path, "a") { |f| f.puts(JSON.generate(entry)) } if @log_path
  end

  # Trả [kết quả text hoặc input của terminal tool, session_id]
  def run_agent(agent, prompt, session_tag: nil)
    session = "#{agent}##{@sessions[agent] += 1}#{session_tag ? "-#{session_tag}" : ''}"
    cfg = ToolConfig::AGENTS.fetch(agent)
    allowed = ToolConfig.allowed_names(agent)
    messages = [{ role: "user", content: prompt }]

    (1..MAX_TURNS).each do |turn|
      payload = build_payload(agent, turn, messages)
      resp = @client.create(payload, agent: agent, session: session)
      uses = resp["content"].select { |b| b["type"] == "tool_use" }
      record(agent: agent, session: session, turn: turn, tool_choice: payload[:tool_choice],
             tools_offered: allowed, stop_reason: resp["stop_reason"], tool_calls: uses.map { |u| u["name"] })
      messages << { role: "assistant", content: resp["content"] }

      if resp["stop_reason"] != "tool_use"
        return [resp["content"].select { |b| b["type"] == "text" }.map { |b| b["text"] }.join, session]
      end

      terminal_input = nil
      results = uses.map do |u|
        content, is_err = dispatch(agent, session, allowed, u)
        if is_err && allowed.include?(u["name"])
          record(type: "tool_error", agent: agent, session: session, tool: u["name"], error: content)
        end
        terminal_input = u["input"] if u["name"] == cfg[:terminal_tool] && !is_err
        { type: "tool_result", tool_use_id: u["id"], content: JSON.generate(content), is_error: is_err }
      end
      messages << { role: "user", content: results }
      return [terminal_input, session] if terminal_input
    end
    raise "#{session} vượt #{MAX_TURNS} lượt"
  end

  def dispatch(agent, session, allowed, use)
    name = use["name"]
    unless allowed.include?(name)
      @violations << { agent: agent, session: session, tool: name }
      record(type: "violation", agent: agent, session: session, tool: name)
      return [{ errorCategory: "permission", isRetryable: false,
                description: "Tool '#{name}' không thuộc bộ tool của #{agent}. Dùng tool của bạn hoặc trả việc về coordinator." }, true]
    end
    if name == "Agent"
      sub = use["input"]["agent_type"]
      tag = use["input"]["prompt"].start_with?("Kiểm chứng") ? "verify" : nil
      out, _sid = run_agent(sub, use["input"]["prompt"], session_tag: tag)
      return [out.is_a?(String) ? { result: out } : { report: out }, false]
    end
    @world.execute(name, use["input"])
  end

  def run(question = QUESTION)
    final, = run_agent("coordinator", question)
    final
  end
end

def verify_log(runner)
  calls = runner.log.reject { |e| e[:type] }
  checks = []
  add = ->(name, ok, detail = "") { checks << { name: name, ok: ok, detail: detail } }

  used = Hash.new { |h, k| h[k] = [] }
  calls.each { |e| used[e[:agent]].concat(e[:tool_calls]) }
  outside = used.flat_map { |a, names| (names.uniq - ToolConfig.allowed_names(a)).map { |n| "#{a}→#{n}" } }
  add.("mỗi agent chỉ dùng tool trong bộ của mình", outside.empty?, outside.empty? ? "OK" : outside.join(", "))

  doc_sessions = calls.select { |e| e[:agent] == "doc_analysis" }.group_by { |e| e[:session] }
  first_ok = doc_sessions.all? do |_s, es|
    f = es.min_by { |e| e[:turn] }
    f[:tool_choice] == { type: "tool", name: "extract_metadata" } && f[:tool_calls] == ["extract_metadata"] &&
      es.reject { |e| e[:turn] == 1 }.all? { |e| e[:tool_choice] == { type: "auto" } }
  end
  add.("doc_analysis: lượt 1 forced extract_metadata, các lượt sau auto (#{doc_sessions.size} phiên)", first_ok)

  syn = used["synthesis"]
  add.("synthesis dùng compile_report + verify_fact", syn.include?("compile_report") && syn.include?("verify_fact"),
       syn.tally.map { |k, v| "#{k}×#{v}" }.join(", "))
  add.("synthesis luôn chạy với tool_choice any",
       calls.select { |e| e[:agent] == "synthesis" }.all? { |e| e[:tool_choice] == { type: "any" } })

  routed = doc_sessions.keys.any? { |s| s.end_with?("-verify") } && used["doc_analysis"].include?("verify_claim")
  add.("nhận định phức tạp được chuyển coordinator → doc_analysis verify_claim", routed)
  add.("không có vi phạm audit", runner.violations.empty?, "#{runner.violations.size} vi phạm")
  [checks, used]
end
